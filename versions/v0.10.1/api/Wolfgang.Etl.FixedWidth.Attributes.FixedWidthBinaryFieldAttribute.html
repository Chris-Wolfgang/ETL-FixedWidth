Marks a property as a field in a fixed-length binary record (#21) — the mainframe counterpart to FixedWidthFieldAttribute. Widths are in bytes (not characters), the byte offset is summed from the ByteLength of preceding columns in Index order, and Type selects how the bytes are decoded (text, COMP-3 packed decimal, or COMP binary integer). Read a record with FixedWidthBinaryExtractor<TRecord>.

[AttributeUsage(AttributeTargets.Property, AllowMultiple = false, Inherited = true)]

Examples

public class AccountRecord
{
    [FixedWidthBinaryField(0, 8, BinaryFieldType.Text)]
    public string AccountId { get; set; } = string.Empty;

    [FixedWidthBinaryField(1, 4, BinaryFieldType.Binary)]
    public int TransactionCount { get; set; }

    [FixedWidthBinaryField(2, 5, BinaryFieldType.PackedDecimal, Scale = 2)]
    public decimal Balance { get; set; }   // PIC S9(7)V99 COMP-3
}
